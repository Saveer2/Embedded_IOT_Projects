#include <SPI.h>
#include <nRF24L01.h>
#include <RF24.h>
#include <Wire.h>
#include <LiquidCrystal_I2C.h>

// NRF24L01 pins
RF24 radio(9, 10); // CE → D9, CSN → D10
const byte address[6] = "00001";

// LCD setup (I2C address 0x27, 16 columns x 2 rows)
LiquidCrystal_I2C lcd(0x27, 16, 2);

void setup() {
  Serial.begin(9600);
  lcd.init();
  lcd.backlight();
  lcd.setCursor(0, 0);
  lcd.print("Receiver Starting");

  // Initialize NRF
  if (!radio.begin()) {
    Serial.println("NRF FAIL");
    lcd.setCursor(0, 1);
    lcd.print("NRF FAIL");
    while (1);
  }

  radio.openReadingPipe(0, address);
  radio.setPALevel(RF24_PA_HIGH); // HIGH is safer for stable reception
  radio.startListening();

  Serial.println("Receiver ready");
  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("Receiver Ready");
}

void loop() {
  if (radio.available()) {
    char text[32] = {0};
    radio.read(&text, sizeof(text));

    Serial.print("Received: ");
    Serial.println(text);

    // Show on LCD
    lcd.clear();
    lcd.setCursor(0, 0);
    lcd.print("Received:");
    lcd.setCursor(0, 1);
    lcd.print(text);
  }
}
