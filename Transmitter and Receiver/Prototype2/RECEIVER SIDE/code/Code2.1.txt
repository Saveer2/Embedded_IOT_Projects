#include <SPI.h>
#include <nRF24L01.h>
#include <RF24.h>
#include <Wire.h>
#include <LiquidCrystal_I2C.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>
#include <DHT.h>

// ---------------- NRF ----------------
RF24 radio(9, 10);   // CE, CSN
const byte address[6] = "00001";

// ---------------- LCD ----------------
LiquidCrystal_I2C lcd(0x27, 16, 2);

// ---------------- OLED ----------------
#define SCREEN_WIDTH 128
#define SCREEN_HEIGHT 64
Adafruit_SSD1306 oled(SCREEN_WIDTH, SCREEN_HEIGHT, &Wire, -1);

// ---------------- DHT ----------------
#define DHTPIN 2
#define DHTTYPE DHT11      // change to DHT22 if needed
DHT dht(DHTPIN, DHTTYPE);

void setup() {
  Serial.begin(9600);

  // LCD init
  lcd.init();
  lcd.backlight();
  lcd.print("Receiver Starting");

  // OLED init
  if (!oled.begin(SSD1306_SWITCHCAPVCC, 0x3C)) {
    Serial.println("OLED FAIL");
    while (1);
  }

  oled.clearDisplay();
  oled.setTextSize(1);
  oled.setTextColor(WHITE);
  oled.setCursor(0, 0);
  oled.println("OLED Ready");
  oled.display();

  // DHT init
  dht.begin();

  // NRF init
  if (!radio.begin()) {
    lcd.setCursor(0, 1);
    lcd.print("NRF FAIL");
    while (1);
  }

  radio.openReadingPipe(0, address);
  radio.setPALevel(RF24_PA_HIGH);
  radio.startListening();

  lcd.clear();
  lcd.print("Receiver Ready");
}

void loop() {
  // ---------- NRF DATA ----------
  if (radio.available()) {
    char text[32] = {0};
    radio.read(&text, sizeof(text));

    Serial.print("Received: ");
    Serial.println(text);

    lcd.clear();
    lcd.setCursor(0, 0);
    lcd.print("Received:");
    lcd.setCursor(0, 1);
    lcd.print(text);
  }

  // ---------- TEMP & HUMIDITY ----------
  float temp = dht.readTemperature();
  float hum = dht.readHumidity();

  oled.clearDisplay();
  oled.setCursor(0, 0);
  oled.println("Environment Data");

  oled.setCursor(0, 20);
  oled.print("Temp: ");
  oled.print(temp);
  oled.println(" C");

  oled.setCursor(0, 35);
  oled.print("Humidity: ");
  oled.print(hum);
  oled.println(" %");

  oled.display();

  delay(2000);
}
